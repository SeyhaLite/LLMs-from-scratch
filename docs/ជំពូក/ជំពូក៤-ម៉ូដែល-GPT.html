ជំពូក ៤៖ ការអនុវត្តម៉ូដែល GPT ពីមូលដ្ឋាន

Note

ប្រភព៖ ថត ch04 និងតារាងមាតិកាក្នុង README.md របស់ upstream។

ជំពូកនេះជាជំហានសំខាន់ — ការផ្គុំបំណែកទាំងអស់ចេញជាម៉ូដែល GPT ពេញលេញ ដែលអនុវត្តពីមូលដ្ឋានដោយប្រើ PyTorch។ ក្រៅពីកូដមេ ជំពូកនេះមានសម្ភារៈបន្ថែម ច្រើនជាងគេអំពីជម្រើសស្ថាបត្យកម្ម Attention និងការវិភាគប្រសិទ្ធភាព។

មេរៀនពេញលេញ (សិក្សាងាយ)

ការពន្យល់ខាងក្រោមជា អត្ថបទដើមរបស់សហគមន៍ខ្មែរ គ្របដណ្តប់គំនិតទាំងអស់នៃជំពូក ៤។ ជំពូកនេះយក multi-head attention ពីជំពូក ៣ មកផ្គុំជាម៉ូដែល GPT ពេញលេញ។

១. ផែនទីស្ថាបត្យកម្មទាំងមូល

២. LayerNorm — ស្ថិរភាពនៃការបណ្តុះបណ្តាល

បញ្ហា៖ នៅពេលទិន្នន័យឆ្លងកាត់ស្រទាប់ច្រើន ការចែកចាយ (distribution) នៃតម្លៃអាចរសាត់ (mean ឃ្លាតពី ០ variance ប្រែប្រួល) ធ្វើឱ្យការបណ្តុះបណ្តាលមិនមានស្ថិរភាព។

ដំណោះស្រាយ៖ LayerNorm ធ្វើឱ្យវ៉ិចទ័រនីមួយៗមាន mean ≈ ០ និង variance ≈ ១ រួចគុណ/បូកជាមួយ parameter ដែលរៀនបាន (gamma scale និង beta shift)។

y = gamma * (x - mean(x)) / sqrt(var(x) + eps) + beta

ចំណាំ៖ LayerNorm ធ្វើ normalization តាមលក្ខណៈនៃ sample នីមួយៗ (across features) ខុសពី BatchNorm ដែលធ្វើតាម batch។ នេះសំខាន់ណាស់សម្រាប់ NLP ព្រោះប្រវែង sequence អាចខុសគ្នា និង batch អាចតូច។

៣. GELU — activation function

GPT ប្រើ GELU (Gaussian Error Linear Unit) ជំនួស ReLU៖

GELU(x) = x * Phi(x)      ដែល Phi = cumulative distribution នៃ Gaussian ធម្មតា

ភាពខុសគ្នាសំខាន់ពី ReLU៖ GELU មិនកាត់តម្លៃអវិជ្ជមានទាំងស្រុងទេ — វាអនុញ្ញាតឱ្យ តម្លៃអវិជ្ជមានតូចៗឆ្លងកាត់បាន ដែលធ្វើឱ្យ gradient រលូនជាង និងជួយការបណ្តុះបណ្តាល។

៤. FeedForward network

បន្ទាប់ពី attention ម៉ូដែលត្រូវការស្រទាប់ដែល ដំណើរការព័ត៌មានលើ token នីមួយៗដាច់ដោយឡែក។ នោះជាតួនាទីរបស់ FeedForward៖

x -> Linear(emb_dim -> 4 * emb_dim) -> GELU -> Linear(4 * emb_dim -> emb_dim) -> Dropout

ចំណុចគួរចាំ៖ កត្តាពង្រីក 4x គឺជាទម្លាប់ស្តង់ដារ។ ស្រទាប់កណ្តាលធំជាង អនុញ្ញាតឱ្យ ម៉ូដែលរៀនអនុគមន៍មិនលីនេអ៊ែរស្មុគស្មាញ។ FeedForward មាន parameter ច្រើនជាង attention ក្នុង Transformer block មួយ។

៥. Shortcut connections (residual connections)

បញ្ហា៖ បណ្តាញជ្រៅខ្លាំងធ្វើឱ្យ gradient រលាយបាត់ (vanishing gradient) ពេល backpropagation។

ដំណោះស្រាយ៖ បូក input ដើមចូលទៅក្នុង output នៃស្រទាប់រង៖

output = x + Sublayer(Norm(x))

អត្ថប្រយោជន៍៖

  • gradient អាចហូរត្រឡប់តាម "ផ្លូវកាត់" ដោយផ្ទាល់ ដូច្នេះម៉ូដែលជ្រៅនៅតែរៀនបាន។

  • ស្រទាប់នីមួយៗគ្រាន់តែរៀន ផលសង (delta) លើ input មិនមែនរៀនតំណាងទាំងស្រុងទេ។

៦. TransformerBlock — ឯកតាដែលផ្គុំឡើងវិញបាន

TransformerBlock គឺជាឯកតាដែលយកអ្វីៗខាងលើមកផ្គុំគ្នា ហើយត្រូវបាន ដាក់ជួរគ្នា N ដង។

ធាតុផ្សំ

តួនាទី

FeedForward

៧. ការគណនាទំហំ attention

អ្នកត្រូវចេះប៉ាន់ប្រមាណការគណនា និងអង្គចងចាំ៖

ចំនួន parameter attention ក្នុង 1 block ~ 4 * emb_dim^2
  (W_q, W_k, W_v, W_o)

attention matrix មានទំហំ  context_length x context_length
  -> ការគណនា attention កើនជា O(n^2) ធៀបនឹងប្រវែង sequence

អង្គចងចាំ KV cache ~ 2 * n_layers * n_heads * d_k * seq_len * batch * 4 bytes (float32)

នេះជាមូលហេតុដែល context length វែង ថ្លៃខ្លាំង និងជាមូលហេតុដែលមានស្ថាបត្យកម្ម attention ជំនួស (GQA, MLA, SWA, DeltaNet) ក្នុងថត bonus នៃជំពូកនេះ។

៨. ការបង្កើតអត្ថបទ (text generation)

ម៉ូដែលដែលផ្គុំរួចបញ្ចេញ logits (ពិន្ទុឆៅសម្រាប់ token នីមួយៗក្នុង vocabulary)។ ដើម្បីបង្កើតអត្ថបទ យើងធ្វើរង្វិលជុំ autoregressive៖

  1. ដាក់ prompt ចូលម៉ូដែល → ទទួល logits សម្រាប់ទីតាំងចុងក្រោយ។

  2. បំលែង logits ទៅជាប្រូបាបដោយ softmax។

  3. ជ្រើសរើស token មួយ (តាមយុទ្ធសាស្ត្រ decoding)។

  4. បន្ថែម token នោះទៅ input រួចធ្វើម្តងទៀតពីជំហាន ១។

យុទ្ធសាស្ត្រជ្រើសរើស token៖

យុទ្ធសាស្ត្រ

លក្ខណៈ

Greedy

យក token ដែលមានប្រូបាបខ្ពស់បំផុតជានិច្ច — ងាយជាប់គាំងក្នុងរង្វិលដដែលៗ

Temperature scaling

ចែក logits នឹង T។ T < 1 ធ្វើឱ្យការចែកចាយមុតស្រួច (អត្ថបទច្បាស់លាស់ ប៉ុន្តែអាចជាន់ដដែល)។ T > 1 ធ្វើឱ្យរាបស្មើ (ច្នៃប្រឌិត ប៉ុន្តែអាចមិនសមហេតុផល)

Top-k

រក្សាតែ token ដែលមានប្រូបាបខ្ពស់បំផុត k ដំបូង រួចលុបផ្សេងទៀតចោល មុនពេលជ្រើសរើស។ ការពារម៉ូដែលមិនឱ្យជ្រើសពាក្យគ្មានន័យ

៩. KV cache — បង្កើនល្បឿន inference

បញ្ហា៖ នៅជំហានបង្កើត token ទី t យើងគណនា K និង V សម្រាប់ token ទាំងអស់ពី ១ ដល់ t ឡើងវិញ ទាំងដែល K/V នៃ token មុនៗ មិនប្រែប្រួល ទេ។

ដំណោះស្រាយ៖ រក្សាទុក K និង V ដែលគណនារួច ហើយគណនាតែ token ថ្មីមួយប៉ុណ្ណោះ។

គ្មាន KV cache:  O(t^2) ការគណនាសរុបសម្រាប់បង្កើត t tokens
មាន KV cache:  O(t)   ប៉ុន្តែត្រូវចំណាយអង្គចងចាំរក្សាទុក K, V

នេះជាអ្វីដែលថត 03_kv-cache អនុវត្ត។ ការផ្លាស់ប្តូរនេះធ្វើឱ្យ inference លឿនជាងច្រើន ប៉ុន្តែត្រូវការអង្គចងចាំបន្ថែម — នេះជា trade-off ដ៏សំខាន់បំផុតមួយក្នុងការបម្រើ LLM។
